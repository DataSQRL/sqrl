>>>flink-sql-no-functions.sql
CREATE TABLE `Clicks` (
  `userid` BIGINT NOT NULL,
  `url` STRING NOT NULL,
  `event_time` TIMESTAMP_LTZ(3) NOT NULL METADATA FROM 'timestamp',
  WATERMARK FOR `event_time` AS `event_time` - INTERVAL '0.0' SECOND
)
WITH (
  'connector' = 'kafka-safe',
  'flexible-json.timestamp-format.standard' = 'ISO-8601',
  'format' = 'flexible-json',
  'properties.auto.offset.reset' = 'earliest',
  'properties.bootstrap.servers' = '${KAFKA_BOOTSTRAP_SERVERS}',
  'properties.compression.type' = 'zstd',
  'properties.group.id' = '${KAFKA_GROUP_ID}',
  'topic' = 'acme-mutation-Clicks'
);
CREATE VIEW `ClicksPerUser`
AS
SELECT `userid`, COUNT(*) AS `clicks`
FROM `Clicks`
GROUP BY `userid`;
CREATE VIEW `HomepageClick`
AS
SELECT *
FROM `Clicks`
WHERE `url` = '/';
CREATE TABLE `ClicksPerUser_1` (
  `userid` BIGINT NOT NULL,
  `clicks` BIGINT NOT NULL,
  PRIMARY KEY (`userid`) NOT ENFORCED
)
WITH (
  'connector' = 'jdbc-sqrl',
  'driver' = 'org.postgresql.Driver',
  'password' = '${POSTGRES_PASSWORD}',
  'table-name' = 'ClicksPerUser',
  'url' = 'jdbc:postgresql://${POSTGRES_AUTHORITY}',
  'username' = '${POSTGRES_USERNAME}'
);
CREATE TABLE `HomepageClick_2` (
  `userid` BIGINT NOT NULL,
  `url` VARCHAR(2147483647) CHARACTER SET `UTF-16LE` NOT NULL,
  `event_time` TIMESTAMP(3) WITH LOCAL TIME ZONE NOT NULL
)
WITH (
  'connector' = 'kafka-safe',
  'flexible-json.timestamp-format.standard' = 'ISO-8601',
  'format' = 'flexible-json',
  'properties.bootstrap.servers' = '${KAFKA_BOOTSTRAP_SERVERS}',
  'properties.compression.type' = 'zstd',
  'properties.group.id' = '${KAFKA_GROUP_ID}',
  'topic' = 'acme-subscription-HomepageClick'
);
EXECUTE STATEMENT SET BEGIN
INSERT INTO `default_catalog`.`default_database`.`ClicksPerUser_1`
SELECT *
FROM `default_catalog`.`default_database`.`ClicksPerUser`
ON CONFLICT DO DEDUPLICATE
;
INSERT INTO `default_catalog`.`default_database`.`HomepageClick_2`
SELECT *
FROM `default_catalog`.`default_database`.`HomepageClick`
;
END
