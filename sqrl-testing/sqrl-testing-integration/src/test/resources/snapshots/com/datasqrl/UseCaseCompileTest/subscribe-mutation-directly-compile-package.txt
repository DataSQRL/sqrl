CREATE TABLE `Events` (
  `event_id` STRING NOT NULL,
  `customer_id` STRING NOT NULL,
  `item` STRING NOT NULL,
  `event_time` TIMESTAMP_LTZ(3) NOT NULL METADATA FROM 'timestamp',
  WATERMARK FOR `event_time` AS `event_time` - INTERVAL '0.0' SECOND
)
WITH (
  'connector' = 'kafka-safe',
  'flexible-json.timestamp-format.standard' = 'ISO-8601',
  'format' = 'flexible-json',
  'properties.auto.offset.reset' = 'earliest',
  'properties.bootstrap.servers' = '${KAFKA_BOOTSTRAP_SERVERS}',
  'properties.compression.type' = 'zstd',
  'properties.group.id' = '${KAFKA_GROUP_ID}',
  'topic' = 'Events'
);
CREATE VIEW `_EventsCopy`
AS
SELECT *
FROM `Events`
WHERE `item` IS NOT NULL;
CREATE VIEW `EventCount`
AS
SELECT COUNT(*) AS `n`
FROM `Events`;
CREATE TABLE `EventCount_1` (
  `n` BIGINT NOT NULL,
  `__pk_hash` INTEGER NOT NULL,
  PRIMARY KEY (`__pk_hash`) NOT ENFORCED
)
WITH (

  'driver' = 'org.postgresql.Driver',

  'table-name' = 'EventCount',
  'url' = 'jdbc:postgresql://${POSTGRES_AUTHORITY}',
  'username' = '${POSTGRES_USERNAME}'
);

  'properties.compression.type' = 'zstd',
  'properties.group.id' = '${KAFKA_GROUP_ID}',
  'topic' = 'Events-subscription'
);
CREATE TABLE `_EventsCopy_3` (
  `event_id` VARCHAR(2147483647) CHARACTER SET `UTF-16LE` NOT NULL,
  `customer_id` VARCHAR(2147483647) CHARACTER SET `UTF-16LE` NOT NULL,
  `item` VARCHAR(2147483647) CHARACTER SET `UTF-16LE` NOT NULL,
  `event_time` TIMESTAMP(3) WITH LOCAL TIME ZONE NOT NULL
)
WITH (
  'connector' = 'kafka-safe',
  'flexible-json.timestamp-format.standard' = 'ISO-8601',
  'format' = 'flexible-json',
  'properties.bootstrap.servers' = '${KAFKA_BOOTSTRAP_SERVERS}',
  'properties.compression.type' = 'zstd',
  'properties.group.id' = '${KAFKA_GROUP_ID}',
  'topic' = '_EventsCopy'
);
EXECUTE STATEMENT SET BEGIN
INSERT INTO `default_catalog`.`default_database`.`EventCount_1`
SELECT `n`, 1 AS `__pk_hash`

ON CONFLICT DO DEDUPLICATE
;
