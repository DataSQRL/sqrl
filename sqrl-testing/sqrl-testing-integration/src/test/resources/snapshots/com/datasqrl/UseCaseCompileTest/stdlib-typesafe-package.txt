CREATE TEMPORARY TABLE `Applications__schema` (
  `application_id` VARCHAR(2147483647) CHARACTER SET `UTF-16LE` NOT NULL,
  `submitted_at` TIMESTAMP(3) WITH LOCAL TIME ZONE NOT NULL,
  `case_file` ROW(`loan` ROW(`amount` DOUBLE, `term_months` BIGINT, `purpose` VARCHAR(2147483647) CHARACTER SET `UTF-16LE`), `applicant` ROW(`employer` VARCHAR(2147483647) CHARACTER SET `UTF-16LE`, `job_title` VARCHAR(2147483647) CHARACTER SET `UTF-16LE`, `stated_annual_income` DOUBLE, `credit_score` BIGINT, `verification_notes` VARCHAR(2147483647) CHARACTER SET `UTF-16LE`), `underwriter_notes` VARCHAR(2147483647) CHARACTER SET `UTF-16LE`)
)
WITH (
  'connector' = 'filesystem',
  'flexible-json.timestamp-format.standard' = 'ISO-8601',
  'format' = 'flexible-json',
  'path' = '${DATA_PATH}/applications.jsonl'
);
CREATE TABLE `Applications` (
  WATERMARK FOR `submitted_at` AS `submitted_at` - INTERVAL '0.001' SECOND
)
WITH (
  'source.monitor-interval' = '1m'
)
LIKE `Applications__schema`;
CREATE VIEW `_UnderwritingInput`
AS
SELECT `application_id`, `submitted_at`, `case_file`, CAST(ROW('Do the verification notes corroborate the stated employment and income?', ROW('Notes confirm employer, role and income', 'Notes contradict or cannot confirm the application')) AS ROW< `instructions` STRING, `criteria` ROW< `true` STRING, `false` STRING > >) AS `q_income_verified`, CASE WHEN `case_file`.`loan`.`purpose` IS NOT NULL THEN CAST(ROW('Which category best describes the purpose of the loan?', ROW('Paying off existing credit cards or loans', 'Repairs, renovation or improvements to a residence', 'Purchase or repair of a vehicle', 'Starting or funding a business', CAST(NULL AS STRING))) AS ROW< `instructions` STRING, `criteria` ROW< `debt_consolidation` STRING, `home_improvement` STRING, `vehicle` STRING, `business` STRING, `other` STRING > >) ELSE NULL END AS `q_purpose`, CAST(ROW('How much underwriting scrutiny does this application warrant?', ARRAY['No concerns', 'Minor concerns', 'Significant concerns', 'Serious red flags']) AS ROW< `instructions` STRING, `criteria` ARRAY< STRING > >) AS `q_scrutiny`
FROM `Applications`;
CREATE VIEW `UnderwritingJudgments`
AS
SELECT `application_id`, CAST(`submitted_at` AS TIMESTAMP_LTZ(3)) AS `submitted_at`, `q_income_verified_answer` AS `income_verified`, `q_purpose_answer`.`choice` AS `purpose`, `q_purpose_answer`.`confidence` AS `purpose_confidence`, `q_purpose_answer`.`probabilities`.`business` AS `business_probability`, `q_scrutiny_answer`.`score` AS `scrutiny`, `q_scrutiny_answer`.`probabilities` AS `scrutiny_probabilities`, `_model` AS `model`, `_error` AS `error`, `rowtime`
FROM TABLE(`typesafe_classify`(`input` => (TABLE `_UnderwritingInput`), `state` => DESCRIPTOR(`case_file`), `noul` => DESCRIPTOR(`q_income_verified`), `choice` => DESCRIPTOR(`q_purpose`), `score` => DESCRIPTOR(`q_scrutiny`), `model` => 'jev-1.13.0', `on_error` => 'NULL', `on_time` => DESCRIPTOR(`submitted_at`)));
CREATE VIEW `DailyUnderwriting`
AS
SELECT `window_start`, `window_end`, COUNT(*) AS `applications`, COUNT(`error`) AS `errors`, SUM(CASE WHEN `income_verified` > 0.5 THEN 1 ELSE 0 END) AS `verified_income`
FROM TABLE(TUMBLE(TABLE `UnderwritingJudgments`, DESCRIPTOR(`rowtime`), INTERVAL '1' DAY))
GROUP BY `window_start`, `window_end`;
