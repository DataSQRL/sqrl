CREATE VIEW `Orders`
AS
SELECT `order_id`, `customer_id`
FROM (VALUES ROW('O1', 'C1'),
   ROW('O2', 'C2')) AS `t` (`order_id`, `customer_id`);
CREATE VIEW `AllOrders`
AS
SELECT `order_id`, `customer_id`
FROM `Orders`;
CREATE TABLE `ItemRequests` (
  `customer_id` STRING NOT NULL,
  `item` STRING NOT NULL,
  `requested_at` TIMESTAMP_LTZ(3) NOT NULL METADATA FROM 'timestamp',
  WATERMARK FOR `requested_at` AS `requested_at` - INTERVAL '0.0' SECOND
)
WITH (
  'connector' = 'kafka-safe',
  'flexible-json.timestamp-format.standard' = 'ISO-8601',
  'format' = 'flexible-json',
  'properties.auto.offset.reset' = 'earliest',
  'properties.bootstrap.servers' = '${KAFKA_BOOTSTRAP_SERVERS}',
  'properties.compression.type' = 'zstd',
  'properties.group.id' = '${KAFKA_GROUP_ID}',
  'topic' = 'ItemRequests'
);
CREATE VIEW `RequestCount`
AS
SELECT COUNT(*) AS `n`
FROM `ItemRequests`;
CREATE TABLE `AllOrders_1` (
  `order_id` CHAR(2) CHARACTER SET `UTF-16LE` NOT NULL,

)

)
WITH (
  'connector' = 'jdbc-sqrl',
  'driver' = 'org.postgresql.Driver',
  'password' = '${POSTGRES_PASSWORD}',
  'table-name' = 'Orders',
  'url' = 'jdbc:postgresql://${POSTGRES_AUTHORITY}',
  'username' = '${POSTGRES_USERNAME}'
);
CREATE TABLE `RequestCount_3` (
  `n` BIGINT NOT NULL,
  `__pk_hash` INTEGER NOT NULL,
  PRIMARY KEY (`__pk_hash`) NOT ENFORCED
)
WITH (
  'connector' = 'jdbc-sqrl',
  'driver' = 'org.postgresql.Driver',
  'password' = '${POSTGRES_PASSWORD}',
  'table-name' = 'RequestCount',
  'url' = 'jdbc:postgresql://${POSTGRES_AUTHORITY}',

);
EXECUTE STATEMENT SET BEGIN

SELECT *
FROM `default_catalog`.`default_database`.`AllOrders`
ON CONFLICT DO DEDUPLICATE
;
INSERT INTO `default_catalog`.`default_database`.`Orders_2`
SELECT *

ON CONFLICT DO DEDUPLICATE
;
INSERT INTO `default_catalog`.`default_database`.`RequestCount_3`
