CREATE VIEW `Sales`
AS
SELECT *
FROM (VALUES ROW('north', 'widget', 120.5),
   ROW('south', 'gadget', 80.0)) AS `t` (`region`, `product`, `amount`);
CREATE TABLE `sales_1` (
  `region` CHAR(5) CHARACTER SET `UTF-16LE` NOT NULL,
  `product` CHAR(6) CHARACTER SET `UTF-16LE` NOT NULL,
  `amount` DECIMAL(4, 1) NOT NULL,
  PRIMARY KEY (`region`) NOT ENFORCED
)
WITH (
  'connector' = 'print',
  'print-identifier' = 'sales'
);
CREATE TABLE `Sales_2` (
  `region` CHAR(5) CHARACTER SET `UTF-16LE` NOT NULL,
  `product` CHAR(6) CHARACTER SET `UTF-16LE` NOT NULL,
  `amount` DECIMAL(4, 1) NOT NULL,
  PRIMARY KEY (`region`) NOT ENFORCED
)
