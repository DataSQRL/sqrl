CREATE TABLE IF NOT EXISTS "Posts" ("id" BIGINT NOT NULL, "title" TEXT NOT NULL, "body" TEXT, "epoch_timestamp" BIGINT NOT NULL, "createdAt" TIMESTAMP WITH TIME ZONE NOT NULL, "__pk_hash" TEXT, PRIMARY KEY ("__pk_hash"));

CREATE INDEX IF NOT EXISTS "Articles_text_c1" ON "Articles" USING GIN (to_tsvector('english', coalesce("title", '') ));
CREATE INDEX IF NOT EXISTS "Posts_text_c1c2" ON "Posts" USING GIN (to_tsvector('english', coalesce("title", '') || ' ' || coalesce("body", '') ))
