FROM (SELECT `id`, `first_name`, `last_name`, `email`, `phone`, `address`, `date_of_birth`, `updated_at`, ROW_NUMBER() OVER (PARTITION BY `id` ORDER BY `updated_at` DESC NULLS LAST) AS `__sqrlinternal_rownum`
  FROM `default_catalog`.`default_database`.`CustomersStream`) AS `t`
WHERE `__sqrlinternal_rownum` = 1;

AS

FROM `Customers`
WHERE `email` IS NOT NULL;

ALTER VIEW `ApplicationsStream`
AS
SELECT `id`, `customer_id`, `loan_type_id`, `amount`, `duration`, `application_date`, `updated_at`, `amount` / 1000 AS `amount_thousands`
FROM (SELECT *
  FROM `sources`.`ApplicationsStream`);
CREATE VIEW `Applications`
AS
SELECT `id`, `customer_id`, `loan_type_id`, `amount`, `duration`, `application_date`, `updated_at`, `amount_thousands`
FROM (SELECT `id`, `customer_id`, `loan_type_id`, `amount`, `duration`, `application_date`, `updated_at`, `amount_thousands`, ROW_NUMBER() OVER (PARTITION BY `id` ORDER BY `updated_at` DESC NULLS LAST) AS `__sqrlinternal_rownum`
  FROM `default_catalog`.`default_database`.`ApplicationsStream`) AS `t`
WHERE `__sqrlinternal_rownum` = 1;
CREATE VIEW `LargeApplications`
AS
SELECT *
FROM `Applications`
WHERE `amount` > 10000 AND `customer_id` IN (SELECT `id`
   FROM `Customers`
   WHERE `last_name` IS NOT NULL);
CREATE VIEW `_ApplicationsByCustomer`
AS
SELECT `a`.`loan_type_id`, `a`.`amount`, `c`.`id` AS `customer_id`
FROM `Applications` AS `a`
 INNER JOIN `Customers` AS `c` ON `a`.`customer_id` = `c`.`id`;
CREATE VIEW `LoanTotals`
AS
SELECT `loan_type_id`, SUM(`amount`) AS `total_amount`, COUNT(*) AS `num_loans`
FROM `_ApplicationsByCustomer`
GROUP BY `loan_type_id`;
CREATE VIEW `_CustomerNameCounts`
AS
SELECT `first_name`, COUNT(*) AS `num_customers`
FROM `Customers`
GROUP BY `first_name`;
CREATE VIEW `CustomerNameCounts`
AS
SELECT `first_name`, `num_customers`
FROM `_CustomerNameCounts`;
CREATE VIEW `CustomersByJurisdiction`
AS
SELECT `address` AS `jurisdiction`, COUNT(*) AS `num_customers`
FROM `Customers`
GROUP BY `address`;
