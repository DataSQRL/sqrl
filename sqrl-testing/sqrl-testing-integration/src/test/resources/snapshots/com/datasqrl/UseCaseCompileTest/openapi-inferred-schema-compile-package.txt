CREATE VIEW `Greeting`
AS
SELECT *
FROM (VALUES ROW(1, 'hello')) AS `t` (`id`, `message`);
CREATE TABLE `greeting_1` (
  `id` INTEGER NOT NULL,
  `message` CHAR(5) CHARACTER SET `UTF-16LE` NOT NULL,
  `__pk_hash` INTEGER NOT NULL,
  PRIMARY KEY (`__pk_hash`) NOT ENFORCED
)
WITH (
  'connector' = 'print',
  'print-identifier' = 'greeting'
);
CREATE TABLE `Greeting_2` (
  `id` INTEGER NOT NULL,
  `message` CHAR(5) CHARACTER SET `UTF-16LE` NOT NULL,
  `__pk_hash` INTEGER NOT NULL,
  PRIMARY KEY (`__pk_hash`) NOT ENFORCED
)
WITH (
  'connector' = 'jdbc-sqrl',
  'driver' = 'org.postgresql.Driver',
  'password' = '${POSTGRES_PASSWORD}',
  'table-name' = 'Greeting',
  'url' = 'jdbc:postgresql://${POSTGRES_AUTHORITY}',
  'username' = '${POSTGRES_USERNAME}'
);
EXECUTE STATEMENT SET BEGIN
INSERT INTO `default_catalog`.`default_database`.`greeting_1`
SELECT `id`, `message`, 1 AS `__pk_hash`
FROM `default_catalog`.`default_database`.`Greeting`
;
INSERT INTO `default_catalog`.`default_database`.`Greeting_2`
SELECT `id`, `message`, 1 AS `__pk_hash`
FROM `default_catalog`.`default_database`.`Greeting`
ON CONFLICT DO DEDUPLICATE
;
END
